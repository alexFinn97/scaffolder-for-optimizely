using System.Threading.Tasks;
using Contoso.Web.Models.Pages;
using EPiServer.Web.Mvc;
using Microsoft.AspNetCore.Mvc;

namespace Contoso.Web.Components;

public class StandardPagePartialViewComponent : AsyncPartialContentComponent<StandardPage>
{
    protected override async Task<IViewComponentResult> InvokeComponentAsync(StandardPage currentContent)
    {
        return await Task.FromResult(View("~/Views/Shared/Partials/StandardPage.cshtml", currentContent));
    }
}
