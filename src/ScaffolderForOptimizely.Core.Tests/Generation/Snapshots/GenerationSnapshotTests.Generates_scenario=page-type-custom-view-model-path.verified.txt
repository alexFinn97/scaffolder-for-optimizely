=== Controllers/ArticlePageController.cs (embedded)
using Contoso.Web.Features.Article;
using Contoso.Web.Models.Pages;
using EPiServer.Web.Mvc;
using Microsoft.AspNetCore.Mvc;

namespace Contoso.Web.Controllers;

public class ArticlePageController : PageController<ArticlePage>
{
    public IActionResult Index(ArticlePage currentPage)
    {
        var model = new ArticleModel(currentPage);

        return View(model);
    }
}

=== Features/Article/ArticleModel.cs (embedded)
using Contoso.Web.Models.Pages;

namespace Contoso.Web.Features.Article;

public class ArticleModel
{
    public ArticleModel(ArticlePage currentPage)
    {
        CurrentPage = currentPage;
    }

    public ArticlePage CurrentPage { get; }
}
