using System.Collections.Generic;
using System.Linq;
using EPiServer.Commerce.Marketing;
using EPiServer.Commerce.Marketing.Extensions;
using EPiServer.Core;

namespace Contoso.Web.Business.Promotions;

/// <summary>
/// Decides whether a cart qualifies for <see cref="CheaperShippingPromotion"/> and what it saves.
/// Found by the promotion engine automatically.
/// </summary>
public class CheaperShippingPromotionProcessor : ShippingPromotionProcessorBase<CheaperShippingPromotion>
{
    private readonly FulfillmentEvaluator _fulfillmentEvaluator;

    public CheaperShippingPromotionProcessor(FulfillmentEvaluator fulfillmentEvaluator)
    {
        _fulfillmentEvaluator = fulfillmentEvaluator;
    }

    protected override FulfillmentStatus GetFulfillmentStatus(CheaperShippingPromotion promotionData, PromotionProcessorContext context) =>
        promotionData.Condition is null
            ? FulfillmentStatus.NotFulfilled
            : promotionData.Condition.GetFulfillmentStatus(context, _fulfillmentEvaluator);

    protected override RewardDescription CreateRewardDescription(
        CheaperShippingPromotion promotionData,
        FulfillmentStatus fulfillmentStatus,
        IEnumerable<RedemptionDescription> redemptions,
        PromotionProcessorContext context) =>
        RewardDescription.CreateMoneyOrPercentageRewardDescription(
            fulfillmentStatus, redemptions, promotionData, promotionData.Discount, context.OrderGroup.Currency, promotionData.Description);

    protected override PromotionItems GetPromotionItems(CheaperShippingPromotion promotionData) =>
        new PromotionItems(promotionData, AllItems(), AllItems());

    private static CatalogItemSelection AllItems() =>
        new CatalogItemSelection(Enumerable.Empty<ContentReference>(), CatalogItemSelectionType.All, true);
}
