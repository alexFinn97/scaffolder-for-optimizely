using System.Collections.Generic;
using System.Linq;
using EPiServer.Commerce.Marketing;
using EPiServer.Commerce.Marketing.Extensions;
using EPiServer.Core;

namespace Contoso.Web.Business.Promotions;

/// <summary>
/// Decides whether a cart qualifies for <see cref="SpendOverPromotion"/> and what it saves.
/// Found by the promotion engine automatically.
/// </summary>
public class SpendOverPromotionProcessor : OrderPromotionProcessorBase<SpendOverPromotion>
{
    private readonly FulfillmentEvaluator _fulfillmentEvaluator;

    public SpendOverPromotionProcessor(FulfillmentEvaluator fulfillmentEvaluator)
    {
        _fulfillmentEvaluator = fulfillmentEvaluator;
    }

    protected override RewardDescription Evaluate(SpendOverPromotion promotionData, PromotionProcessorContext context)
    {
        if (promotionData.Condition is null || promotionData.Discount is null)
        {
            return RewardDescription.CreateNotFulfilledDescription(promotionData, FulfillmentStatus.NotFulfilled);
        }

        var status = promotionData.Condition.GetFulfillmentStatus(context, _fulfillmentEvaluator);
        if (status != FulfillmentStatus.Fulfilled)
        {
            return RewardDescription.CreateNotFulfilledDescription(promotionData, status);
        }

        var redemptions = new[] { CreateRedemptionDescription(context.OrderForm) };
        return RewardDescription.CreateMoneyOrPercentageRewardDescription(
            status, redemptions, promotionData, promotionData.Discount, context.OrderGroup.Currency, promotionData.Description);
    }

    protected override PromotionItems GetPromotionItems(SpendOverPromotion promotionData) =>
        new PromotionItems(promotionData, AllItems(), AllItems());

    private static CatalogItemSelection AllItems() =>
        new CatalogItemSelection(Enumerable.Empty<ContentReference>(), CatalogItemSelectionType.All, true);
}
