using System.Collections.Generic;
using System.Linq;
using EPiServer.Commerce.Marketing;
using EPiServer.Core;

namespace Contoso.Web.Business.Promotions;

/// <summary>
/// Decides whether a cart qualifies for <see cref="BuyThreeGetDiscountPromotion"/> and what it saves.
/// Found by the promotion engine automatically.
/// </summary>
public class BuyThreeGetDiscountPromotionProcessor : EntryPromotionProcessorBase<BuyThreeGetDiscountPromotion>
{
    private readonly CollectionTargetEvaluator _targetEvaluator;
    private readonly FulfillmentEvaluator _fulfillmentEvaluator;

    public BuyThreeGetDiscountPromotionProcessor(
        CollectionTargetEvaluator targetEvaluator,
        FulfillmentEvaluator fulfillmentEvaluator,
        RedemptionDescriptionFactory redemptionDescriptionFactory)
        : base(redemptionDescriptionFactory)
    {
        _targetEvaluator = targetEvaluator;
        _fulfillmentEvaluator = fulfillmentEvaluator;
    }

    protected override RewardDescription Evaluate(BuyThreeGetDiscountPromotion promotionData, PromotionProcessorContext context)
    {
        if (promotionData.Condition is null || promotionData.DiscountTargets is null || promotionData.Discount is null)
        {
            return RewardDescription.CreateNotFulfilledDescription(promotionData, FulfillmentStatus.NotFulfilled);
        }

        var lineItems = GetLineItems(context.OrderForm).ToList();

        // Does the cart hold enough of the condition items?
        var conditionCodes = _targetEvaluator.GetApplicableCodes(lineItems, promotionData.Condition.Items, promotionData.Condition.MatchRecursive);
        var status = _fulfillmentEvaluator.GetStatusForBuyQuantityPromotion(conditionCodes, lineItems, promotionData.Condition.RequiredQuantity, 0);
        if (status != FulfillmentStatus.Fulfilled)
        {
            return RewardDescription.CreateNotFulfilledDescription(promotionData, status);
        }

        // Discount the target items in the cart.
        var targetCodes = _targetEvaluator.GetApplicableCodes(lineItems, promotionData.DiscountTargets.Items, promotionData.DiscountTargets.MatchRecursive);
        var redemptions = GetRedemptions(promotionData, context, targetCodes);
        return RewardDescription.CreateMoneyOrPercentageRewardDescription(
            status, redemptions, promotionData, promotionData.Discount, context.OrderGroup.Currency, promotionData.Description);
    }

    protected override PromotionItems GetPromotionItems(BuyThreeGetDiscountPromotion promotionData) =>
        new PromotionItems(
            promotionData,
            new CatalogItemSelection(promotionData.Condition?.Items ?? new List<ContentReference>(), CatalogItemSelectionType.Specific, promotionData.Condition?.MatchRecursive ?? true),
            new CatalogItemSelection(promotionData.DiscountTargets?.Items ?? new List<ContentReference>(), CatalogItemSelectionType.Specific, promotionData.DiscountTargets?.MatchRecursive ?? true));
}
